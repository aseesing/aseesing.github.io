uint64_t decimal_length(uint64_t v) {
const auto zeros = std::countl_zero(v);
return kLengthByLeadingZeros.base[zeros]
+ (v >= kLengthByLeadingZeros.threshold[zeros]);
}
 
size_t u64_to_chars(uint64_t value, char* buf) {
const uint64_t length = decimal_length(value);
// top = value / 10^16 (version 6: not shown)
// value / 10^8, beside it (version 6: not shown)
// high4 = 4 × the 8 digits below top (version 7: not shown)
// low4 = 4 × the last 8 digits (version 7: not shown)
// top_length, from a table (version 6: not shown)
// zeros to drop, from a table (version 6: not shown)
// store the top characters at buf (version 8: not shown)
const __m128i drop = _mm_loadu_si128(kDropLeading + zeros);
// store _mm_shuffle_epi8(digits16(high4, low4), drop)
// at buf + top_length (version 8: not shown)
return length;
}
 
__m128i digits16(uint64_t high4, uint64_t low4) {
const __m256i x = _mm256_inserti128_si256(
_mm256_castsi128_si256(_mm_cvtsi64_si128(high4)),
_mm_cvtsi64_si128(low4), 1);
// q = the first four digits of each half
// (version 7: not shown)
const __m256i rq = _mm256_add_epi64(x,
_mm256_mul_epu32(q, kSplit));
const __m256i v = _mm256_shuffle_epi8(rq, kSpread);
const __m256i quotients = _mm256_mulhi_epu16(
_mm256_mulhi_epu16(v, kRecip), kShift);
const __m256i tens = _mm256_mullo_epi16(
_mm256_slli_epi64(quotients, 16), kTimesTen);
const __m256i digits = _mm256_sub_epi16(
_mm256_add_epi16(quotients, kAsciiZero), tens);
return _mm_packus_epi16(_mm256_castsi256_si128(digits),
_mm256_extracti128_si256(digits, 1));
}
